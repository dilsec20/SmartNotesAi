CA2 Project Report: SmartNotes AI

Student Name: Dilip  |  Platform: Android (Kotlin + Jetpack Compose)

1. Project Overview

SmartNotes AI is an AI-powered Android study assistant that helps students create notes, scan documents (PDF/Image) using OCR, generate AI-powered summaries and MCQs, and ask follow-up questions about their study material. The entire application is built in a single-file architecture (MainActivity.kt) for maximum readability and ease of explanation. It uses Google Gemini AI for intelligent text processing and Google ML Kit for on-device text extraction.

2. Module-Wise Breakdown

  • GUI Module: Jetpack Compose UI with enhanced cards, color theming, chat bubbles, and responsive layout. Uses simple enum-based screen navigation.
  • Data Persistence Module: Room Database for local CRUD operations on study notes with support for storing AI-generated summaries and MCQs.
  • ML & AI Engine Module: Google ML Kit for OCR text extraction from images/PDFs. Google Gemini AI (gemini-3.1-flash-lite) for summary generation, MCQ creation, and interactive follow-up Q&A via chat sessions.

3. Functionalities

  • Notes Management: Create, view, and delete text-based study notes with enhanced card UI.
  • Document Scanner: Upload images or PDFs, extract text via ML Kit OCR, and save as notes.
  • AI Summary: Generate concise summaries with key bullet points from note content or scanned documents.
  • AI MCQ Generator: Auto-generate 5-question multiple choice quizzes for self-assessment.
  • Follow-up Chat: Ask clarifying questions after summary generation; AI retains context via Gemini chat sessions.
  • Save Scanned Results: Save extracted text along with generated summary and MCQs as a new note.

// Room Database

@Entity(tableName = "notes")
data class Note(
    @PrimaryKey(autoGenerate = true) val id: Int = 0,
    val title: String = "",
    val content: String = "",
    val summary: String = "",
    val mcqs: String = "",
    val createdAt: Long = System.currentTimeMillis()
)

@Dao
interface NoteDao {
    @Query("SELECT * FROM notes ORDER BY createdAt DESC")
    fun getAllNotes(): Flow<List<Note>>
    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertNote(note: Note): Long
    @Delete
    suspend fun deleteNote(note: Note)
}

@Database(entities = [Note::class], version = 3, exportSchema = false)
abstract class NoteDatabase : RoomDatabase() {
    abstract fun noteDao(): NoteDao
    companion object {
        @Volatile private var INSTANCE: NoteDatabase? = null
        fun getDatabase(context: Context): NoteDatabase {
            return INSTANCE ?: synchronized(this) {
                Room.databaseBuilder(context.applicationContext, NoteDatabase::class.java, "ainote_db")
                    .fallbackToDestructiveMigration(dropAllTables = true).build().also { INSTANCE = it }
            }
        }
    }
}

data class ChatMsg(val role: String, val text: String)

// ViewModel

class MainViewModel(application: Application) : AndroidViewModel(application) {
    private val noteDao = NoteDatabase.getDatabase(application).noteDao()
    val allNotes = noteDao.getAllNotes()

    private val generativeModel = GenerativeModel(
        modelName = "gemini-3.1-flash-lite",
        apiKey = "YOUR_GEMINI_API_KEY"
    )
    private var chatSession = generativeModel.startChat()
    private val textRecognizer = TextRecognition.getClient(TextRecognizerOptions.DEFAULT_OPTIONS)

    private fun clean(text: String): String {
        return text.replace(Regex("""\$([^$]+)\$"""), "$1")
            .replace("**", "").replace("###", "").replace("##", "").replace("#", "")
            .replace("`", "").lines().joinToString("\n") { it.trimEnd() }
    }

    // StateFlows for extracted text, summary, mcqs, processing state, chat history
    private val _extractedText = MutableStateFlow("")
    private val _summary = MutableStateFlow("")
    private val _mcqs = MutableStateFlow("")
    private val _isProcessing = MutableStateFlow(false)
    private val _chatHistory = MutableStateFlow<List<ChatMsg>>(emptyList())

    fun saveNote(title: String, content: String, summary: String = "", mcqs: String = "") {
        viewModelScope.launch {
            noteDao.insertNote(Note(title = title, content = content, summary = summary, mcqs = mcqs))
        }
    }

    fun processImage(context: Context, uri: Uri) {
        viewModelScope.launch {
            _isProcessing.value = true
            val image = InputImage.fromFilePath(context, uri)
            val result = textRecognizer.process(image).await()
            _extractedText.value = result.text
            _isProcessing.value = false
        }
    }

    fun processPdf(context: Context, uri: Uri) {
        viewModelScope.launch {
            _isProcessing.value = true
            val fd = context.contentResolver.openFileDescriptor(uri, "r")
            if (fd != null) {
                val renderer = PdfRenderer(fd)
                val sb = StringBuilder()
                for (i in 0 until renderer.pageCount) {
                    val page = renderer.openPage(i)
                    val bmp = Bitmap.createBitmap(page.width, page.height, Bitmap.Config.ARGB_8888)
                    page.render(bmp, null, null, PdfRenderer.Page.RENDER_MODE_FOR_DISPLAY)
                    page.close()
                    val result = textRecognizer.process(InputImage.fromBitmap(bmp, 0)).await()
                    sb.append(result.text).append("\n")
                }
                renderer.close(); fd.close()
                _extractedText.value = sb.toString()
            }
            _isProcessing.value = false
        }
    }

    fun generateSummary(text: String) {
        viewModelScope.launch {
            _isProcessing.value = true
            chatSession = generativeModel.startChat()
            val prompt = "Summarize with key bullet points, plain text only, no markdown:\n$text"
            val response = chatSession.sendMessage(prompt)
            _summary.value = clean(response.text ?: "Could not generate summary.")
            _chatHistory.value = listOf(ChatMsg("model", _summary.value))
            _isProcessing.value = false
        }
    }

    fun generateMCQs(text: String) {
        viewModelScope.launch {
            _isProcessing.value = true
            val response = generativeModel.generateContent("Generate 5 MCQs, plain text only:\n$text")
            _mcqs.value = clean(response.text ?: "Could not generate MCQs.")
            _isProcessing.value = false
        }
    }

    fun sendFollowUp(question: String) {
        viewModelScope.launch {
            _chatHistory.value = _chatHistory.value + ChatMsg("user", question)
            _isProcessing.value = true
            val response = chatSession.sendMessage(content("user") { text(question) })
            _chatHistory.value = _chatHistory.value + ChatMsg("model", clean(response.text ?: ""))
            _isProcessing.value = false
        }
    }
}

enum class Screen { NoteList, AddNote, Scanner, NoteDetail }

@Composable
fun MainApp() {
    var screen by remember { mutableStateOf(Screen.NoteList) }
    var selectedNote by remember { mutableStateOf<Note?>(null) }
    val vm: MainViewModel = viewModel()
    when (screen) {
        Screen.NoteList -> NoteListScreen(vm) { s, n -> selectedNote = n; screen = s }
        Screen.AddNote -> AddNoteScreen(vm) { screen = Screen.NoteList }
        Screen.Scanner -> ScannerScreen(vm) { screen = Screen.NoteList }
        Screen.NoteDetail -> NoteDetailScreen(vm, selectedNote) { screen = Screen.NoteList }
    }
}
